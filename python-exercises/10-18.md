## Q17 GIL: threads vs processes, pick correctly

```python
"""
TASK:
  Two workloads below: one CPU-bound, one I/O-bound (simulated with sleep,
  standing in for a network/disk call). You're given a threaded and a
  multiprocess runner for each but they're mismatched on purpose.

  1. Run all four combinations and time them.
  2. Explain, in terms of the GIL, why threading helps one workload and not
     the other, and why multiprocessing has the opposite tradeoff.
  3. Fix the mismatch: pair each workload with the execution model that
     actually helps, and justify with numbers, not just theory.
"""

import time
import threading
import multiprocessing


def cpu_bound_work(n):
    total = 0
    for i in range(n):
        total += i * i
    return total


def io_bound_work(seconds):
    time.sleep(seconds)
    return seconds


def run_threaded(fn, args_list):
    threads = [threading.Thread(target=fn, args=(a,)) for a in args_list]
    start = time.perf_counter()
    for t in threads: t.start()
    for t in threads: t.join()
    return time.perf_counter() - start


def run_multiprocess(fn, args_list):
    processes = [multiprocessing.Process(target=fn, args=(a,)) for a in args_list]
    start = time.perf_counter()
    for p in processes: p.start()
    for p in processes: p.join()
    return time.perf_counter() - start


if __name__ == "__main__":
    # mismatched on purpose -- CPU work run threaded, I/O work run multiprocess
    cpu_args = [5_000_000] * 4
    io_args = [1] * 4

    t1 = run_threaded(cpu_bound_work, cpu_args)
    print(f"CPU-bound threaded: {t1:.2f}s")

    t2 = run_multiprocess(io_bound_work, io_args)
    print(f"I/O-bound multiprocess: {t2:.2f}s")
```
